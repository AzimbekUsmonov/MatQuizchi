// =====================================================
//  FIREBASE FIRESTORE SECURITY RULES
//  Firebase Console -> Firestore -> Rules ga ko'chiring
// =====================================================

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // ─── HELPER FUNCTIONS ───────────────────────────
    function isAuth() {
      return request.auth != null;
    }

    function isAdmin() {
      // Asosiy admin ID (har doim kirish huquqi bor)
      // Qo'shimcha adminlar Firestore settings/admins ga qo'shiladi
      return isAuth() && (
        request.auth.uid == '8377539642' ||
        (
          exists(/databases/$(database)/documents/settings/admins) &&
          request.auth.uid in get(/databases/$(database)/documents/settings/admins).data.ids
        )
      );
    }

    function isOwner(uid) {
      return isAuth() && request.auth.uid == uid;
    }

    // ─── USERS ──────────────────────────────────────
    match /users/{userId} {
      // O'z profilini o'qish va yozish
      allow read: if isAuth();
      allow create: if isOwner(userId);
      allow update: if isOwner(userId) || isAdmin();
      allow delete: if isAdmin();
    }

    // ─── QUESTIONS ──────────────────────────────────
    match /questions/{qId} {
      // Barcha autentifikatsiya qilinganlar o'qiy oladi
      allow read: if isAuth();
      // Faqat admin yozadi
      allow write: if isAdmin();
    }

    // ─── TOPICS ─────────────────────────────────────
    match /topics/{tId} {
      allow read: if isAuth();
      allow write: if isAdmin();
    }

    // ─── QUIZ HISTORY ────────────────────────────────
    match /quizHistory/{hId} {
      allow read: if isAuth();
      allow create: if isAuth() && request.resource.data.uid == request.auth.uid;
      allow update, delete: if isAdmin();
    }

    // ─── DAILY USAGE ────────────────────────────────
    match /dailyUsage/{key} {
      allow read, write: if isAuth();
    }

    // ─── SPEED STATS ────────────────────────────────
    match /speedStats/{uid} {
      allow read: if isAuth();
      allow write: if isOwner(uid) || isAdmin();
    }

    // ─── PVP BATTLES ────────────────────────────────
    match /pvpBattles/{battleId} {
      allow read: if isAuth();
      allow create: if isAuth();
      allow update: if isAuth() &&
        (resource.data.player1 == request.auth.uid ||
         resource.data.player2 == request.auth.uid ||
         resource.data.player2 == null);
      allow delete: if isAdmin();
    }

    // ─── CLANS ──────────────────────────────────────
    match /clans/{clanId} {
      allow read: if isAuth();
      allow create: if isAuth();
      allow update: if isAuth() &&
        (resource.data.leaderId == request.auth.uid || isAdmin());
      allow delete: if isAdmin();
    }

    // ─── TOURNAMENTS ────────────────────────────────
    match /tournaments/{tId} {
      allow read: if isAuth();
      allow write: if isAdmin();

      match /matches/{mId} {
        allow read: if isAuth();
        allow write: if isAdmin();
      }
    }

    // ─── BROADCASTS ─────────────────────────────────
    match /broadcasts/{bId} {
      allow read: if isAdmin();
      allow write: if isAdmin();
    }

    // ─── SEASON HISTORY ─────────────────────────────
    match /seasonHistory/{sId} {
      allow read: if isAuth();
      allow write: if isAdmin();
    }

    // ─── SETTINGS ───────────────────────────────────
    match /settings/{doc} {
      allow read: if isAuth();
      allow write: if isAdmin();
    }
  }
}
