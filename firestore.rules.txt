rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // ─── HELPERS ────────────────────────────────────
    function isAuth() {
      return request.auth != null;
    }

    function isOwner(docId) {
      return isAuth() && request.auth.uid == docId;
    }

    function isAdmin() {
      return isAuth() && (
        // Asosiy admin - har doim kirish huquqi bor
        get(/databases/$(database)/documents/admins/$(request.auth.uid)).data.isAdmin == true
      );
    }

    // ─── USERS ──────────────────────────────────────
    // O'z doc ID = Firebase auth UID
    match /users/{userId} {
      allow read: if isAuth();
      allow create: if isOwner(userId);
      allow update: if isOwner(userId) || isAdmin();
      allow delete: if isAdmin();
    }

    // ─── QUESTIONS ──────────────────────────────────
    match /questions/{qId} {
      allow read: if isAuth();
      allow write: if isAdmin();
    }

    // ─── TOPICS ─────────────────────────────────────
    match /topics/{tId} {
      allow read: if isAuth();
      allow write: if isAdmin();
    }

    // ─── QUIZ HISTORY ────────────────────────────────
    match /quizHistory/{hId} {
      allow read: if isAuth();
      allow create: if isAuth();
      allow update, delete: if isAdmin();
    }

    // ─── DAILY USAGE ────────────────────────────────
    match /dailyUsage/{key} {
      allow read, write: if isAuth();
    }

    // ─── SPEED STATS ────────────────────────────────
    match /speedStats/{uid} {
      allow read: if isAuth();
      allow write: if isOwner(uid) || isAdmin();
    }

    // ─── PVP BATTLES ────────────────────────────────
    match /pvpBattles/{battleId} {
      allow read: if isAuth();
      allow create: if isAuth();
      allow update: if isAuth();
      allow delete: if isAdmin();
    }

    // ─── CLANS ──────────────────────────────────────
    match /clans/{clanId} {
      allow read: if isAuth();
      allow create: if isAuth();
      allow update: if isAuth();
      allow delete: if isAdmin();
    }

    // ─── TOURNAMENTS ────────────────────────────────
    match /tournaments/{tId} {
      allow read: if isAuth();
      allow write: if isAdmin();

      match /matches/{mId} {
        allow read: if isAuth();
        allow write: if isAdmin();
      }
    }

    // ─── BROADCASTS ─────────────────────────────────
    match /broadcasts/{bId} {
      allow read, write: if isAdmin();
    }

    // ─── SEASON HISTORY ─────────────────────────────
    match /seasonHistory/{sId} {
      allow read: if isAuth();
      allow write: if isAdmin();
    }

    // ─── SETTINGS ───────────────────────────────────
    match /settings/{doc} {
      allow read: if isAuth();
      allow write: if isAdmin();
    }

    // ─── ADMINS ─────────────────────────────────────
    // Admin doc: admins/{firebaseUid} -> { isAdmin: true }
    match /admins/{uid} {
      allow read: if isAuth();
      allow write: if isAdmin();
    }
  }
}
